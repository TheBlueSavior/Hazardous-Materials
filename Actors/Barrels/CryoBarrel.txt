Class HM_CryoBarrel : Actor
{
	int timer;
	Default
	{
		Health 400;
		Radius 10;
		Height 32;
		+SOLID
		+SHOOTABLE
		+ACTIVATEMCROSS
		+DONTGIB
		+NOICEDEATH
		+OLDRADIUSDMG
		+NOBLOODDECALS
		DeathSound "";
		Bloodtype "HM_NormalHitSpark";
		Obituary "$OB_BARREL";
	}
	
	override void PostBeginPlay()
	{
	Super.PostBeginPlay();
	tex[0] = TexMan.CheckForTexture("DUSTC0",TexMan.Type_Sprite);
	tex[1] = TexMan.CheckForTexture("GNSMI0",TexMan.Type_Sprite);
	tex[2] = TexMan.CheckForTexture("DTPRB0",TexMan.Type_Sprite);
	}
 
	TextureID tex[3];
	
	States
	{
	Spawn:
		TNT1 A 0 NoDelay { timer = random(80,165); }
	SpawnLoop:
		BRL_ C 6;
		BRL_ C 0 A_JumpIfHealthLower(375,"Burning");
		BRL_ C 6
		{
		A_SpawnParticleEx
			( "",tex[1], style: STYLE_Normal, flags: SPF_RELATIVE|SPF_ROLL|SPF_NEGATIVE_FADESTEP,
			lifetime: random(200,250), size: random(10,30), xoff: random(-15,15), yoff: random(-15,15), zoff: random(20,60),velx: frandom(-0.1,0.1), vely: frandom(-0.1,0.1), velz: frandom(-0.3,-0.1),
			startalphaf: frandom(0.02,0.05), fadestepf: frandom(-0.001,-0.0005), sizestep: frandom(0,0.5), startroll: random(-180,180)
			);
		}
		BRL_ C 0 A_JumpIfHealthLower(375,"Burning");
		Loop;
	Burning:
		TNT1 A 0;
		TNT1 A 0 A_StartSound("HMBarrel/Break",38,CHANF_OVERLAP,1);
		TNT1 A 0 A_StartSound("HMBarrel/Strain",33,CHANF_OVERLAP,frandom(0.7,1.0));
		TNT1 A 0 A_StartSound("Hiss/Idle",32,volume:0.85);
		TNT1 AAA 0 A_SpawnItemEx("HM_BarrelShrapnel", 0, 0, 20,frandom(-6, 6),frandom(-6, 6),frandom(4, 14),frandom(0, 360), SXF_NOCHECKPOSITION|SXF_ABSOLUTEVELOCITY);
		TNT1 AAAAA 0 A_SpawnItemEx("HM_BarrelShockwave",0,0,30,0,0,0,0,SXF_NOCHECKPOSITION,150);
	BurningLoop:
		BRL_ D 1 Light("HML_CryoLightA")
		{
		A_SpawnParticleEx("",tex[2],style: STYLE_Add,flags: SPF_RELATIVE|SPF_FULLBRIGHT,lifetime: 4,size: frandom(50,180),xoff: random(-2,2),yoff:random(-2,2),30,startalphaf: frandom(0.1,0.6));
		A_SpawnParticleEx
			( "",tex[1], style: STYLE_Normal, flags: SPF_RELATIVE|SPF_ROLL,
			lifetime: random(60,120), size: 6, xoff: 0, yoff: 0, zoff: 25,velx: frandom(-1,1), vely: frandom(-1,1), velz: frandom(5,10),
			accelz: frandom(-0.4,-0.1), startalphaf: 0.85, fadestepf: -1, sizestep: frandom(2,4), startroll: random(-180,180)
			);
		timer--;
		}
		BRL_ D 0
		{
			if(timer < 0)
			SetStateLabel("Boom");
		}
		Loop;
	Boom:
		TNT1 A 0;	
		TNT1 A 0 A_Die;
		Goto Death;
	Death:
		TNT1 A 0;
		TNT1 A 0 A_NoBlocking();
		TNT1 A 0 A_StartSound("HMBarrel/Strain",33,CHANF_OVERLAP,frandom(0.7,1.0));
		TNT1 A 0 A_StartSound("Cryo/Freeze",34,CHANF_OVERLAP);
		TNT1 A 0 A_StartSound("Cryo/Shatter",35,CHANF_OVERLAP);
		TNT1 A 0 A_RadiusGive("HM_CryoFreezing",300,RGF_PLAYERS|RGF_MONSTERS,1);
		TNT1 AAAAAAAAAAAAAA 0 A_SpawnItemEx ("IceChunk",0,0,random(2,30),random(-4,4),random(-4,4),random(-4,4),0,SXF_NOCHECKPOSITION,0);
		TNT1 AAAAAAAAA 0 A_SpawnItemEx("HM_BarrelShrapnel", 0, 0, random(5,15), frandom(-5, 5),frandom(-5, 5),frandom(0, 6),frandom(0, 360), SXF_NOCHECKPOSITION|SXF_ABSOLUTEVELOCITY);
		TNT1 AAAAAAAA 0 A_SpawnItemEx("HM_CryoFrostCloud2",0,0,random(2,35),frandom(-2,2),frandom(-2,2),frandom(-1,0),0,0,50);
		TNT1 A 0 A_StopSound(32);
		BRL_ EEEEEEEEEEEEEEEEEEEEEEEEEEEEEEEEEEEEEEEEEEEEEEEEEEEEEEEEEEEEEEEEEEEEEEEEEEEEEEEEEEEEEEEEEEEE 5 Light("HML_CryoLightB")
			{
			A_SpawnItemEx("HM_CryoFrostCloud",0,0,random(2,10),frandom(-2,2),frandom(-2,2),frandom(0,0),0,0,120);
			A_SpawnItemEx("HM_CryoFrostCloud",0,0,random(2,10),frandom(-2,2),frandom(-2,2),frandom(0,0),0,0,120);
			A_SpawnItemEx("HM_SmallWhiteExplosionFlames",0,0,10,frandom(-1,1),frandom(-1,1),frandom(0,2),0,SXF_NOCHECKPOSITION);
				A_SpawnParticleEx
				(
				"",
				tex[1],
				style: STYLE_Translucent,
				flags: SPF_ROLL|SPF_RELATIVE,
				lifetime: random(60,500),
				size: 15.0,
				angle: 0,
				xoff: random(100,-100),
				yoff: random(100,-100),
				zoff: random(0,35),
				velx: frandom(-1,1),
				vely: frandom(-1,1),
				velz: 0,
				accelx: frandom(-0.01,0.01),
				accely: frandom(-0.01,0.01),
				accelz: frandom(-0.01,0),
				startalphaf: frandom(0.1,0.7),
				fadestepf: -1,
				sizestep: random(1,3),
				startroll: random(-180,180),
				rollvel: frandom(-1,1),
				rollacc: 0
				);
			}
		BRL_ E -1;
		Stop;
	}
}
